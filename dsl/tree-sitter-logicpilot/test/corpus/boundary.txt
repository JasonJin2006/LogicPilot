==================
Multiple resources then multiple processes
==================

model Shop {
  resource Drill { capacity = 2 }
  resource Lathe { capacity = 4 failure_rate = 0.05 }
  resource Crane { capacity = 1 }
  process A {
    source In { arrival = poisson(3) }
    service Drill { time = normal(8, 1) }
  }
  process B {
    source In { arrival = poisson(1) }
    queue Hold { capacity = 5 }
    service Crane { time = constant(10) }
  }
}

==================
Comments interleaved everywhere
==================

/* header block comment */
model Commented { // model line comment
  /* before resource */
  resource R {
    capacity = 2 // inline
    /* between fields */
    failure_rate = 0.5
  }
  // before process
  process P {
    source S { arrival = poisson(1) /* arrivals */ }
    /* before queue */
    queue Q { capacity = 3 }
    service R { time = exponential(2) } // tail
  }
}
/* footer block comment */
