End-to-end Factory model from dsl-spec
==================

// Factory example — v0 DSL
model Factory {
  resource Machine {
    capacity = 3
    failure_rate = 0.01
  }

  process Production {
    source Order {
      arrival = poisson(5)
    }
    queue Buffer {
      capacity = 50
    }
    service Machine {
      time = normal(10, 2)
    }
  }
}

// M/M/1: single server, Poisson arrivals, exponential service
model MM1 {
  resource Server {
    capacity = 1
    failure_rate = 0.0
  }
  process Flow {
    source Jobs {
      arrival = poisson(1)
    }
    queue WaitingLine {
      capacity = 100
    }
    service Server {
      time = exponential(2)
    }
  }
}
