    source Order {
      arrival = poisson(5)
    }

      capacity = 50
    }
    service Machine {

}

// M/M/1: single server, Poisson arrivals, exponential service
model MM1 {
  resource Server {
    capacity = 1
    failure_rate = 0.0
  }
  process Flow {
    source Jobs {
      arrival = poisson(1)
    }
    queue WaitingLine {
      capacity = 100
    }
    service Server {
      time = exponential(2)
    }
  }
}
