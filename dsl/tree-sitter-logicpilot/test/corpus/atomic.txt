model PulseChain {
  atomic Pulser {
    time_advance = constant(1.0)
    on_timeout { emit pulse }
  }
  atomic Sink {
    state seen = false
    on_input pulse { seen = true }
  }
  couple Pulser.pulse -> Sink.pulse
}

model Machine {
  atomic Cell {
    state busy = false
    state load = 0
    time_advance = exponential(2.0)
    on_input job_in { busy = true, load = 1 }
    on_timeout { busy = false emit done }
  }
}

model Cell {
  atomic Core {
    state phase: int = 0
    time_advance = constant(1.0)
    in job_in: Signal
    out done: Signal
    on_input job_in { phase = 1 }
    on_timeout { phase = 0; emit done }
  }
}
